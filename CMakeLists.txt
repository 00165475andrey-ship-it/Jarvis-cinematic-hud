cmake_minimum_required(VERSION 3.20)
project(JarvisDesktop LANGUAGES CXX)
add_executable(JarvisDesktop WIN32 src/JarvisApp.cpp)
target_compile_features(JarvisDesktop PRIVATE cxx_std_17)
target_compile_definitions(JarvisDesktop PRIVATE UNICODE _UNICODE)
target_link_libraries(JarvisDesktop PRIVATE comctl32 advapi32)
if(MSVC)
  target_compile_options(JarvisDesktop PRIVATE /W4 /permissive- /utf-8)
endif()
